cmake_minimum_required(VERSION 3.21)
project(GitLearning VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

add_library(task_core src/Task.cpp src/Task.h src/TaskManager.cpp src/TaskManager.h)
target_include_directories(task_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(task_core PUBLIC cxx_std_20)

add_executable(git_learning src/main.cpp src/Console.cpp src/Console.h)
target_link_libraries(git_learning PRIVATE task_core)

function(enable_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
enable_warnings(task_core)
enable_warnings(git_learning)

include(CTest)
if(BUILD_TESTING)
    add_executable(task_manager_tests tests/test_task_manager.cpp)
    target_link_libraries(task_manager_tests PRIVATE task_core)
    enable_warnings(task_manager_tests)
    add_test(NAME task_manager COMMAND task_manager_tests)

    add_executable(console_tests tests/test_console.cpp src/Console.cpp src/Console.h)
    target_link_libraries(console_tests PRIVATE task_core)
    enable_warnings(console_tests)
    add_test(NAME console COMMAND console_tests)
    add_test(NAME cli_help COMMAND git_learning --help)
    set_tests_properties(cli_help PROPERTIES PASS_REGULAR_EXPRESSION "Commands:")
    add_test(NAME cli_invalid_option COMMAND git_learning --unknown)
    set_tests_properties(cli_invalid_option PROPERTIES WILL_FAIL TRUE)
endif()
